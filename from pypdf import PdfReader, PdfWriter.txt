from pypdf import PdfReader, PdfWriter
import os

def split_pdf_by_pages(input_path, output_dir, pages_per_split=30):
    reader = PdfReader(input_path)
    total_pages = len(reader.pages)

    os.makedirs(output_dir, exist_ok=True)

    for start in range(0, total_pages, pages_per_split):
        writer = PdfWriter()
        end = min(start + pages_per_split, total_pages)

        for i in range(start, end):
            writer.add_page(reader.pages[i])

        output_file = os.path.join(output_dir, f"split_{start+1:04d}-{end:04d}.pdf")
        with open(output_file, "wb") as f:
            writer.write(f)

    print(f"총 {total_pages}페이지를 {pages_per_split}장씩 분할 완료했습니다.")

# 사용 예시
if __name__ == "__main__":
    input_pdf = input("분할할 PDF 파일 경로를 입력하세요: ")
    output_folder = "output_pdfs"
    split_pdf_by_pages(input_pdf, output_folder)
